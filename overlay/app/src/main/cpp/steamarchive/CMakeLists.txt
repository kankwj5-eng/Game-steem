cmake_minimum_required(VERSION 3.22.1)
project(steamarchive C)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lzma_sdk/C")
set(SDK_FILES Bcj2 Bra Bra86 BraIA64 CpuArch Delta Lzma2Dec LzmaDec Ppmd7 Ppmd7Dec
    7zCrc 7zCrcOpt 7zArcIn 7zBuf 7zBuf2 7zDec 7zFile 7zStream)
set(SDK_SOURCES "")
foreach(FILE IN LISTS SDK_FILES)
    list(APPEND SDK_SOURCES "${SDK}/${FILE}.c")
endforeach()
add_library(steamarchive SHARED steamarchive.c ${SDK_SOURCES})
target_compile_options(steamarchive PRIVATE -O2 -Wall -Wextra)
target_include_directories(steamarchive PRIVATE "${SDK}")
target_compile_definitions(steamarchive PRIVATE Z7_PPMD_SUPPORT Z7_EXTRACT_ONLY _POSIX_C_SOURCE=200809L)
set_target_properties(steamarchive PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_compile_options(steamarchive PRIVATE -O2 -Wall -Wextra)
target_include_directories(steamarchive PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(steamarchive PRIVATE pthread)
endif()
